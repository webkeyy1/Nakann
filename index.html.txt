  <script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-app.js";
    import {
      getAuth, createUserWithEmailAndPassword, signInWithEmailAndPassword,
      signOut, onAuthStateChanged
    } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-auth.js";
    import {
      getFirestore, doc, getDoc, setDoc, updateDoc, addDoc,
      collection, getDocs, query, where, limit, serverTimestamp
    } from "https://www.gstatic.com/firebasejs/10.12.2/firebase-firestore.js";

    const firebaseConfig = {
      apiKey: "AIzaSyA06iDqbxW5JGK3wYTKnG45BiJWvgZC5HU",
      authDomain: "htheanh-real.firebaseapp.com",
      projectId: "htheanh-real",
      storageBucket: "htheanh-real.firebasestorage.app",
      messagingSenderId: "679370583353",
      appId: "1:679370583353:web:dbf4fea310f5e08f370ce8",
      measurementId: "G-296WHXVZ4V"
    };

    const app = initializeApp(firebaseConfig);
    const auth = getAuth(app);
    const db = getFirestore(app);

    let currentUser = null;
    let currentBalance = 0;

    function showMsg(text) {
      const el = document.getElementById("msgBox");
      const err = document.getElementById("errBox");
      err.style.display = "none";
      el.textContent = text;
      el.style.display = "block";
    }

    function showErr(text) {
      const el = document.getElementById("errBox");
      const msg = document.getElementById("msgBox");
      msg.style.display = "none";
      el.textContent = text;
      el.style.display = "block";
    }

    window.registerUser = async function() {
      const email = document.getElementById("emailInput").value.trim();
      const pass = document.getElementById("passInput").value;
      try {
        const cred = await createUserWithEmailAndPassword(auth, email, pass);
        await setDoc(doc(db, "users", cred.user.uid), {
          email: email,
          balance: 0,
          createdAt: serverTimestamp()
        });
        showMsg("Đăng ký thành công!");
      } catch (e) {
        showErr("Lỗi đăng ký: " + e.message);
      }
    };

    window.loginUser = async function() {
      const email = document.getElementById("emailInput").value.trim();
      const pass = document.getElementById("passInput").value;
      try {
        await signInWithEmailAndPassword(auth, email, pass);
        showMsg("Đăng nhập thành công!");
      } catch (e) {
        showErr("Lỗi đăng nhập: " + e.message);
      }
    };

    window.logoutUser = async function() {
      await signOut(auth);
      location.reload();
    };

    onAuthStateChanged(auth, async (user) => {
      currentUser = user;
      const authBox = document.getElementById("authBox");
      const shopArea = document.getElementById("shopArea");
      const userBar = document.getElementById("userBar");

      if (!user) {
        authBox.style.display = "block";
        shopArea.style.display = "none";
        userBar.innerHTML = "Chưa đăng nhập.";
        return;
      }

      authBox.style.display = "none";
      shopArea.style.display = "block";

      const userRef = doc(db, "users", user.uid);
      const snap = await getDoc(userRef);
      if (!snap.exists()) {
        await setDoc(userRef, { email: user.email, balance: 0 });
        currentBalance = 0;
      } else {
        currentBalance = Number(snap.data().balance || 0);
      }

      userBar.innerHTML = `
        <div class="row" style="align-items:center;">
          <div>
            <div><b>Tài khoản:</b> ${user.email}</div>
            <div><b>Số dư:</b> <span style="color:#4ade80; font-size:18px; font-weight:bold;">${currentBalance.toLocaleString("vi-VN")}đ</span></div>
          </div>
          <div style="max-width:120px;">
            <button class="gray" onclick="logoutUser()">Đăng xuất</button>
          </div>
        </div>
      `;

      document.getElementById("transferNote").textContent = "NAP " + user.uid.slice(0, 8).toUpperCase();

      await loadProducts();
      await loadOrders();
    });

    async function loadProducts() {
      const listEl = document.getElementById("productsList");
      const selectEl = document.getElementById("keyProdSelect");
      const snap = await getDocs(collection(db, "products"));

      if (snap.empty) {
        listEl.innerHTML = "Chưa có gói key nào. Hãy thêm ở phần Admin bên dưới.";
        selectEl.innerHTML = `<option value="">-- Chưa có gói --</option>`;
        return;
      }

      let html = "";
      let opts = "";

      for (const d of snap.docs) {
        const p = d.data();
        const price = Number(p.price || 0);

        const stockSnap = await getDocs(
          query(collection(db, "keys"), where("productId", "==", d.id), where("sold", "==", false))
        );
        const stockCount = stockSnap.size;

        html += `
          <div class="item">
            <div style="display:flex; justify-content:space-between; align-items:center;">
              <div>
                <b>${p.name}</b> <span class="badge">Còn ${stockCount} key</span>
                <div style="color:#38bdf8; font-weight:bold; margin-top:4px;">${price.toLocaleString("vi-VN")}đ</div>
              </div>
              <div style="width:110px;">
                <button ${stockCount === 0 ? "disabled style='opacity:0.5'" : ""} onclick="buyKey('${d.id}', '${p.name.replace(/'/g, "")}', ${price})">
                  Mua ngay
                </button>
              </div>
            </div>
          </div>
        `;

        opts += `<option value="${d.id}">${p.name} (${price.toLocaleString("vi-VN")}đ)</option>`;
      }

      listEl.innerHTML = html;
      selectEl.innerHTML = opts;
    }

    window.buyKey = async function(productId, productName, price) {
      if (!currentUser) return showErr("Vui lòng đăng nhập.");
      if (currentBalance < price) {
        return showErr(`Số dư không đủ! Cần ${price.toLocaleString("vi-VN")}đ, bạn đang có ${currentBalance.toLocaleString("vi-VN")}đ.`);
      }

      try {
        const stockQ = query(
          collection(db, "keys"),
          where("productId", "==", productId),
          where("sold", "==", false),
          limit(1)
        );
        const stockSnap = await getDocs(stockQ);

        if (stockSnap.empty) {
          return showErr("Gói này đã hết key trong kho!");
        }

        const keyDoc = stockSnap.docs[0];
        const keyCode = keyDoc.data().code;
        const newBalance = currentBalance - price;

        await updateDoc(doc(db, "keys", keyDoc.id), {
          sold: true,
          buyerId: currentUser.uid,
          buyerEmail: currentUser.email,
          soldAt: serverTimestamp()
        });

        await updateDoc(doc(db, "users", currentUser.uid), {
          balance: newBalance
        });

        await addDoc(collection(db, "orders"), {
          userId: currentUser.uid,
          productId: productId,
          productName: productName,
          price: price,
          keyCode: keyCode,
          createdAt: serverTimestamp()
        });

        currentBalance = newBalance;
        showMsg(`Mua thành công gói "${productName}"! Key của bạn: ${keyCode}`);
        onAuthStateChanged(auth, currentUser);
      } catch (e) {
        showErr("Lỗi khi mua key: " + e.message);
      }
    };

    async function loadOrders() {
      const ordersEl = document.getElementById("ordersList");
      if (!currentUser) return;
      const q = query(collection(db, "orders"), where("userId", "==", currentUser.uid));
      const snap = await getDocs(q);

      if (snap.empty) {
        ordersEl.innerHTML = "Bạn chưa mua key nào.";
        return;
      }

      let html = "";
      snap.forEach((d) => {
        const o = d.data();
        html += `
          <div class="item">
            <div><b>${o.productName}</b> - <span style="color:#cbd5e1;">${Number(o.price || 0).toLocaleString("vi-VN")}đ</span></div>
            <div class="key-box">KEY: ${o.keyCode}</div>
          </div>
        `;
      });
      ordersEl.innerHTML = html;
    }

    window.addProduct = async function() {
      const name = document.getElementById("prodName").value.trim();
      const price = Number(document.getElementById("prodPrice").value);
      if (!name || !price || price <= 0) return showErr("Nhập đủ tên gói và giá hợp lệ.");

      try {
        await addDoc(collection(db, "products"), {
          name: name,
          price: price,
          createdAt: serverTimestamp()
        });
        document.getElementById("prodName").value = "";
        document.getElementById("prodPrice").value = "";
        showMsg("Đã thêm gói hàng mới!");
        await loadProducts();
      } catch (e) {
        showErr("Lỗi thêm gói: " + e.message);
      }
    };

    window.importKeys = async function() {
      const productId = document.getElementById("keyProdSelect").value;
      const raw = document.getElementById("keysTextarea").value;
      const lines = raw.split("\n").map(x => x.trim()).filter(Boolean);

      if (!productId) return showErr("Hãy chọn gói hàng trước.");
      if (lines.length === 0) return showErr("Hãy nhập ít nhất 1 mã key.");

      try {
        for (const k of lines) {
          await addDoc(collection(db, "keys"), {
            productId: productId,
            code: k,
            sold: false,
            createdAt: serverTimestamp()
          });
        }
        document.getElementById("keysTextarea").value = "";
        showMsg(`Đã nhập thành công ${lines.length} key vào kho!`);
        await loadProducts();
      } catch (e) {
        showErr("Lỗi nhập key: " + e.message);
      }
    };
  </script>
